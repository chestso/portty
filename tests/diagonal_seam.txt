# Test script for diagonal box-drawing seam verification
#
# This script tests that diagonal box-drawing characters (U+2571-U+2573)
# render seamlessly when stacked vertically, without 1-pixel gaps at row
# boundaries.
#
# Usage: portty -S tests/diagonal_seam.txt
#
# Diagonal characters in UTF-8:
#   U+2571 (╱) = \xE2\x95\xB1
#   U+2572 (╲) = \xE2\x95\xB2
#   U+2573 (╳) = \xE2\x95\xB3

# Clear screen and move to home
emit \e[2J\e[H

# Display test header
sendln echo Diagonal Box-Drawing Seam Test
sendln echo =============================
sendln echo 

# Test diagonal characters stacked vertically
sendln echo Test 1: U+2571 (forward slash diagonal)
emit-raw \xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1
sendln echo
emit-raw \xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1\xE2\x95\xB1
sendln echo
sendln echo

sendln echo Test 2: U+2572 (backslash diagonal)
emit-raw \xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2
sendln echo
emit-raw \xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2\xE2\x95\xB2
sendln echo
sendln echo

sendln Test 3: U+2573 (crossing diagonals)
emit-raw \xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3
sendln echo
emit-raw \xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3\xE2\x95\xB3
sendln echo
sendln echo

sendln Look for gaps between rows
sendln echo

# Wait for rendering and save screenshot
wait 500
screendump /tmp/diagonal_seam.png
wait 200
quit
